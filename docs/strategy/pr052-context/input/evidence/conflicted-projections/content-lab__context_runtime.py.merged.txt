def scoped_path(value,roots):
    if not isinstance(value,str) or '\0' in value or not Path(value).is_absolute():raise ValueError('ABSOLUTE_PATH_REQUIRED')
    p=Path(value)
    if any(x.is_symlink() for x in (p,*p.parents)):raise ValueError('CONTEXT_LINK_PATH')
    resolved=p.resolve()
    if not any(resolved.is_relative_to(Path(r).resolve()) and resolved!=Path(r).resolve() for r in roots):raise ValueError('PATH_OUTSIDE_OPERATOR_SCOPE')
    return p

def schema_state(store):
    if store is None:return 'UNKNOWN_STORE'
    try:
        db=read_connection(store)
        try:
            tables={r[0] for r in db.execute("SELECT name FROM sqlite_master WHERE type='table'")}
            if 'context_schema' not in tables:return 'SETUP_REQUIRED'
            if [r[0] for r in db.execute('SELECT version FROM context_schema')]!=[1]:return 'UNSUPPORTED_SCHEMA'
            needed={'context_sources','context_raw_parts','context_source_heads','context_intents',
                    'context_intent_fences','context_packets','context_packet_sources','context_selections','core_control'}
            return 'READY_V1' if needed<=tables else 'SETUP_REQUIRED'
        finally:db.close()
    except (OSError,ValueError,__import__('sqlite3').Error):return 'SETUP_REQUIRED'

def submit(store,policy,service,action,args,operation_id):
    c=config(policy,service);identifier(operation_id);validate_arguments(action,args,c)
    if action not in c['actions']:raise ValueError('CONTEXT_ACTION_NOT_GRANTED')
    if schema_state(store)!='READY_V1':raise ValueError('CONTEXT_SETUP_OR_SCHEMA_REQUIRED')
    payload={'kind':'context_service','service':service,'action':action,'arguments':args,

def run_job(core,job):
    p=job['payload'];validate_job(p,core.policy);c=config(core.policy,p['service']);ns=c['namespace'];a=p['arguments'];op=p['operation_id'];action=p['action']
    core.heartbeat(job)
    def tick():core.heartbeat(job)
    # One intent fences immediate local operations too. Multi-part owners have
    # their own resumable intents; jobs reuse the same logical ID after recovery.
    own=action in {'CAPTURE','BUILD_PACKET','PROJECT','ANNOTATE','DELETE','PURGE'}
    if not own:
        with lib.view(core.store,write=True) as db:
            db.execute('BEGIN IMMEDIATE');old=lib.begin_intent(db,ns,op,'service:'+action,a)
            if old is not None:return old
    if action=='CAPTURE':result=lib.import_file(core.store,ns,a['source_key'],a['path'],op,progress=tick)
    elif action=='SELECT':result=packets.create_selection(core.store,ns,a['rows'],parent=a.get('parent'))
    elif action=='REMOVE_RANGE':result=packets.remove_selected_range(core.store,ns,a['selection_id'],a['ordinal'])
    elif action=='ADOPT_ITEM':result={'source_ref':packets.adopt_item_ref(core.store,ns,a['item_id'])}
    elif action=='BUILD_PACKET':result=packets.compile_task_document(core.store,ns,a['selection_id'],a['goal'],a['criteria'],op,part_bytes=a.get('part_bytes',4096),progress=tick)
    elif action=='PROJECT':result=packets.create_share_projection(core.store,ns,a['packet_id'],c['destinations'][a['destination']],op,redactions=a.get('redactions'),progress=tick)
    elif action=='EXPORT':result=packets.export_projection(core.store,ns,a['projection_id'],a['output'],c['destinations'][a['destination']],progress=tick)
    elif action=='RESULT':
        with scoped_path(a['path'],c['input_roots']).open('rb') as f:raw=f.read(128_001)
        result=packets.import_result(core.store,ns,a['result_id'],raw)
    elif action=='NEED_CONTEXT':result=packets.need_context(core.store,ns,a['request'])
    elif action=='ANNOTATE':result=lib.annotate(core.store,ns,a['source_ref'],a['annotation'],op)
    elif action=='DELETE':result=lib.tombstone(core.store,ns,a['source_key'],op)
    elif action=='PURGE':result=recovery.purge_tombstoned_source(core.store,ns,a['source_key'],op)
    elif action=='BACKUP':result=recovery.snapshot_backup(core.store,a['output'],progress=tick)
    elif action=='RESTORE_COPY':result=recovery.restore_to_copy(a['backup'],a['output'],current_store=core.store,progress=tick)
    elif action=='SYNC_EXPORT':result=recovery.export_sync(core.store,ns,a['output'])
    elif action=='SYNC_IMPORT':result=recovery.import_sync(core.store,ns,a['bundle'])
    else:raise ValueError('UNREGISTERED_CONTEXT_HANDLER')
    core.heartbeat(job)
    if not own:
        with lib.view(core.store,write=True) as db:
            db.execute('BEGIN IMMEDIATE');lib.ensure_running(db);lib.finish_intent(db,ns,op,result)
    return result
